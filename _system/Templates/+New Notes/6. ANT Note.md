<%*
// Creates a new ANT capture note in /Notes and opens it.
// Modelled on "1. Create Note" so it works from the Cmd+Shift+N menu (and from a hotkey).
const DESTINATION_FOLDER = "Notes";

// Don't run while editing a template
const currentFolder = tp.file.folder(true);
if (currentFolder.startsWith("_system/Templates")) {
  new Notice("Can't run this here.");
} else {
  // Title: use selected text if present, otherwise prompt
  const selection = tp.file.selection();
  let title = (selection && selection.trim())
    ? selection.trim()
    : await tp.system.prompt("📍 Title new note");

  if (title && title.trim()) {
    // sanitise for a filename
    title = title.trim().replace(/\.$/, "").replace(/[\\/:*?"<>|]/g, "-");

    const newFilePath = `${DESTINATION_FOLDER}/${title}.md`;
    if (app.vault.getAbstractFileByPath(newFilePath)) {
      new Notice("A note with that title already exists.");
    } else {
      // Capture metadata
      const noteAuthor = await tp.system.prompt("Your name (must match your People note)");
      const practice = await tp.system.suggester(
        ["facilitating", "building / making", "participating", "documenting", "observing", "other"],
        ["facilitating", "building / making", "participating", "documenting", "observing", "other"],
        false,
        "From which practice are you writing right now?"
      );
      const session = await tp.system.prompt("Session / moment (e.g. 'Day 1 — schema discussion')");
      const date = tp.date.now("YYYY-MM-DD HH:mm");

      const content = `---
class: Note
title: ${title}
noteAuthor: "[[${noteAuthor ?? ""}]]"
from-practice: ${practice ?? ""}
date: ${date}
session: ${session ?? ""}
case: attribution-workshop
tags: [ant/capture]
---

# ${title}

> **Describe, don't interpret.** Name people and things specifically. Quote the actual words when someone frames what the problem is. Keep the order things happened in. It's fine — good, even — to capture more than you can make sense of yet. Characterisation happens later, with the tool; this note is the raw material.

## What happened
*Concrete, in sequence. Name who/what specifically ("Maria argued…", "the schema draft…"), not "the group discussed the thing". Drop in short direct quotes where they land — especially when someone defines the problem or makes their fix the necessary one.*


## Where it stuck / who disagreed
*Pin the exact sticking point — not "there was disagreement" but what, precisely, was unresolved, and the framing each side used. Sticking points are the good stuff: they're where things are still moving.*


## Artifacts & links
*The actual doc / thread / recording / screenshot this rests on, so it can be traced back to source.*


## Things I'm unsure about
*e.g. "I didn't follow why X mattered to them." A flag for whoever characterises this later not to over-read.*


---
*Optional sensitizing tags — hunches, not categories. Drop inline where they fit:  #controversy  #stuck  #surprise  #black-box*
`;

      if (!app.vault.getAbstractFileByPath(DESTINATION_FOLDER)) {
        await app.vault.createFolder(DESTINATION_FOLDER);
      }
      const newFile = await app.vault.create(newFilePath, content);
      await app.workspace.openLinkText(newFile.path, "", true);

      // If launched from selected text, leave a link behind in the source note
      if (selection && selection.trim()) {
        tR = `[[${title}]]`;
      }
    }
  }
}
%>
